import os from fastapi import FastAPI from fastapi.responses import HTMLResponse from pydantic import BaseModel from dotenv import load_dotenv from groq import Groq load_dotenv() api_key = os.getenv("GROQ_API_KEY") if not api_key: raise ValueError( "GROQ_API_KEY was not found. " "Please add it to your .env file (local) or Vercel project settings (deployed)." ) client = Groq(api_key=api_key) app = FastAPI() class ChatRequest(BaseModel): message: str history: list[dict] | None = None SYSTEM_PROMPT = { "role": "system", "content": "You are a helpful, intelligent AI assistant." } @app.get("/", response_class=HTMLResponse) @app.get("/chat.html", response_class=HTMLResponse) def serve_chat_page(): # Load chat.html from the same folder as this file html_path = os.path.join(os.path.dirname(__file__), "chat.html") with open(html_path, "r", encoding="utf-8") as f: return f.read() @app.post("/chat") def chat(request: ChatRequest): messages = [SYSTEM_PROMPT] if request.history: messages.extend(request.history) messages.append({"role": "user", "content": request.message}) try: response = client.chat.completions.create( model="openai/gpt-oss-120b", messages=messages, temperature=0.7, max_tokens=1024, stream=False, ) reply = response.choices[0].message.content return { "reply": reply } except Exception as e: return {"error": str(e)}
